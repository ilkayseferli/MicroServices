pipeline {
    agent any

    stages {
        stage('Checkout & Clean') {
            steps {
                cleanWs() // Workspace'i temizle
                checkout scm
            }
        }

        stage('Docker Build (No Cache)') {
            steps {
                script {
                    // İmaj isimlerini docker-compose.yml ile TAM uyuşacak şekilde (ilkayseferli/...) veriyoruz
                    // --no-cache ekleyerek ocelot.json değişikliğini zorla içeri alıyoruz
                    
                    sh "docker build --no-cache -t ilkayseferli/apigateway -f APIGateway/Dockerfile ."
                    sh "docker build --no-cache -t ilkayseferli/contactapi -f Services/ContactAPI/Contact.API/Dockerfile ."
                    sh "docker build --no-cache -t ilkayseferli/resapi -f Services/ResAPI/Reservation.API/Dockerfile ."
                }
            }
        }

        stage('Deploy') {
            steps {
                // --build eklemiyoruz çünkü yukarıda manuel ve --no-cache ile yaptık. 
                // Sadece up -d diyerek yereldeki taze imajları ayağa kaldırıyoruz.
                sh 'docker-compose up -d'
            }
        }
    }
    
    post {
        always {
            sh 'docker image prune -f'
        }
    }
}
