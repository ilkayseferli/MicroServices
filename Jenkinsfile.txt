pipeline {
    agent any

    stages {
        stage('Checkout & Clean') {
            steps {
                cleanWs() // Eski dosya kalıntılarını temizle
                checkout scm
            }
        }

        stage('Docker Build (No Cache)') {
            steps {
                script {
                    // İmaj isimlerini docker-compose.yml ile tam eşleştiriyoruz
                    // --no-cache ekleyerek ocelot.json değişikliğini içeriye zorla basıyoruz
                    
                    sh "docker build --no-cache -t ilkayseferli/apigateway -f APIGateway/Dockerfile ."
                    sh "docker build --no-cache -t ilkayseferli/contactapi -f Services/ContactAPI/Contact.API/Dockerfile ."
                    sh "docker build --no-cache -t ilkayseferli/resapi -f Services/ResAPI/Reservation.API/Dockerfile ."
                }
            }
        }

        stage('Deploy') {
            steps {
                // down komutu eski network ve konteyner kalıntılarını temizler
                sh 'docker-compose down'
                sh 'docker-compose up -d'
            }
        }
    }
}
